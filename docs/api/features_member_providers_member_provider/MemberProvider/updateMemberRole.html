Future<bool> updateMemberRole(String userId, UserRole newRole) async {
  final user = _dataService.getUserById(userId);
  if (user == null) return false;

  try {
    await _dataService.updateUser(user.copyWith(role: newRole));
    notifyListeners();
    return true;
  } catch (e) {
    return false;
  }
}
